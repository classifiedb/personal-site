import re

EMAIL = re.compile(r"[\w.+-]+@[\w-]+\.[\w.]+")

def redact(text: str) -> str:
    """Replace email addresses with a placeholder before sending text to an AI service."""
    return EMAIL.sub("[REDACTED_EMAIL]", text)

print(redact("Contact me at someone@example.com"))
